import java.util.Arrays;
import java.util.Scanner;

public class TravelPackages {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter Number of Travel Packages: ");
        int n = sc.nextInt();

        int[] prices = new int[n];

        System.out.println("\nEnter Package Prices:");
        for (int i = 0; i < n; i++) {
            prices[i] = sc.nextInt();
        }

        Arrays.sort(prices);

        System.out.println("\nSorted Travel Package Prices:\n");
        for (int price : prices) {
            System.out.print(price + " ");
        }

        sc.close();
    }
}

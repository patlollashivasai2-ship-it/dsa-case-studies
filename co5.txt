import java.util.Scanner;

public class TravelBenefit {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter number of travel packages: ");
        int n = sc.nextInt();

        int[] cost = new int[n];
        int[] benefit = new int[n];

        System.out.println("Enter package costs:");
        for (int i = 0; i < n; i++) {
            cost[i] = sc.nextInt();
        }

        System.out.println("Enter package benefits:");
        for (int i = 0; i < n; i++) {
            benefit[i] = sc.nextInt();
        }

        System.out.print("Enter budget: ");
        int budget = sc.nextInt();

        int[][] dp = new int[n + 1][budget + 1];

        for (int i = 1; i <= n; i++) {
            for (int w = 0; w <= budget; w++) {
                if (cost[i - 1] <= w) {
                    dp[i][w] = Math.max(
                        benefit[i - 1] + dp[i - 1][w - cost[i - 1]],
                        dp[i - 1][w]
                    );
                } else {
                    dp[i][w] = dp[i - 1][w];
                }
            }
        }

        System.out.println("Maximum Travel Benefit = " + dp[n][budget]);
        sc.close();
    }
}
